cmake_minimum_required(VERSION 3.24)
project(ss2_native VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(SDL3 3.2 REQUIRED CONFIG)
find_package(Vulkan 1.3 REQUIRED)
add_library(ss2_window STATIC src/render/window.cpp)
target_include_directories(ss2_window PUBLIC include src)
target_link_libraries(ss2_window PUBLIC SDL3::SDL3 PRIVATE Vulkan::Vulkan)
target_compile_options(ss2_window PRIVATE -Wall -Wextra -Wpedantic)
add_executable(ss2 src/main.cpp src/ui/tutorial_selection.cpp)
target_link_libraries(ss2 PRIVATE ss2_window)
target_compile_options(ss2 PRIVATE -Wall -Wextra -Wpedantic)
find_package(ZLIB 1.2.9 REQUIRED)
find_package(JPEG REQUIRED)
add_library(ss2_assets STATIC src/assets/sprite.cpp src/assets/mission.cpp src/assets/terrain.cpp
    src/assets/scenery.cpp src/assets/indexed.cpp src/assets/infantry.cpp src/assets/vehicles.cpp src/assets/placement.cpp)
target_include_directories(ss2_assets PUBLIC src)
target_link_libraries(ss2_assets PRIVATE ZLIB::ZLIB JPEG::JPEG)
target_link_libraries(ss2 PRIVATE ss2_assets)
target_compile_options(ss2_assets PRIVATE -Wall -Wextra -Wpedantic)
add_executable(ss2_sprite tools/sprite_probe.cpp)
target_link_libraries(ss2_sprite PRIVATE ss2_assets)
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME sprite_archive COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/sprite_test.py $<TARGET_FILE:ss2_sprite>)
  add_executable(map_test tests/map_test.cpp)
  target_link_libraries(map_test PRIVATE ss2_assets)
  add_test(NAME map_assets COMMAND map_test)
  add_executable(clock_test tests/clock_test.cpp)
  target_include_directories(clock_test PRIVATE include)
  add_test(NAME simulation_independence COMMAND clock_test)
  add_test(NAME cli_help COMMAND ss2 --help)
  add_test(NAME invalid_frame_count COMMAND ${CMAKE_COMMAND} -DPROGRAM=$<TARGET_FILE:ss2>
    "-DARGS=--frames;0" "-DEXPECTED=--frames requires a positive integer" -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_failure.cmake)
  option(SS2_TEST_WINDOW "Present frames during CTest (requires Vulkan WSI)" OFF)
  if(SS2_TEST_WINDOW)
    option(SS2_TEST_VALIDATION "Require Khronos validation in Vulkan tests" OFF)
    if(SS2_TEST_VALIDATION)
      set(validation_arg --validate)
      file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/no-vulkan-layers")
      add_test(NAME missing_validation_layer COMMAND ${CMAKE_COMMAND} -DPROGRAM=$<TARGET_FILE:ss2>
        "-DARGS=--frames;1;--validate" "-DEXPECTED=Required VK_LAYER_KHRONOS_validation is unavailable"
        -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_failure.cmake)
      set_tests_properties(missing_validation_layer PROPERTIES ENVIRONMENT
        "VK_LAYER_PATH=${CMAKE_CURRENT_BINARY_DIR}/no-vulkan-layers;VK_ADD_LAYER_PATH=;VK_INSTANCE_LAYERS=")
    endif()
    add_test(NAME vulkan_present COMMAND ss2 --frames 8 ${validation_arg})
    add_executable(tutorial_selection_test tests/tutorial_selection_test.cpp src/ui/tutorial_selection.cpp)
    target_link_libraries(tutorial_selection_test PRIVATE ss2_window)
    add_test(NAME tutorial_selection COMMAND tutorial_selection_test ${validation_arg})
    add_executable(resize_test tests/resize_test.cpp)
    target_link_libraries(resize_test PRIVATE ss2_window)
    add_test(NAME vulkan_resize COMMAND resize_test ${validation_arg})
    set_tests_properties(vulkan_resize PROPERTIES SKIP_RETURN_CODE 77)
    add_executable(wsi_recovery_test tests/wsi_recovery.cpp)
    target_link_libraries(wsi_recovery_test PRIVATE ss2_window Vulkan::Vulkan ${CMAKE_DL_LIBS})
    add_test(NAME vulkan_wsi_recovery COMMAND wsi_recovery_test ${validation_arg})
    set_tests_properties(tutorial_selection vulkan_present vulkan_resize vulkan_wsi_recovery PROPERTIES TIMEOUT 30
      FAIL_REGULAR_EXPRESSION "Validation Error|VUID-")
  endif()
endif()
